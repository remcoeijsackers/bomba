Variables & values
let x = 42
x += 1
let s = "x is {x}"     # interpolation
let raw = 'no {x} here'
let ok = true and not nil
let xs = [1, 2, 3]
let m = {name: "bomba", (key): 1}

    
Control flow
if x > 10 { ... }
elif x > 5 { ... }
else { ... }

while x > 0 { x -= 1 }
for item in xs { ... }
for i, item in xs { ... }
for key, val in m { ... }
break / continue

    
Functions
fn add(a, b = 1) {
  return a + b
}
let sq = fn(x) => x * x
map(xs, sq)
filter(xs, fn(x) => x > 1)
reduce(xs, fn(a, b) => a + b)
